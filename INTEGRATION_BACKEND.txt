src/main/java/com/energieplus/controllers/

FormulaireController.java
```java
@RestController
@RequestMapping("/api/formulaires")
@CrossOrigin(origins = "http://localhost:5173") // Frontend React
public class FormulaireController {
    
    @Autowired
    private FormulaireService service;
    
    @GetMapping
    public ResponseEntity<List<Formulaire>> getAll() {
        return ResponseEntity.ok(service.findAll());
    }
    
    @GetMapping("/{id}")
    public ResponseEntity<Formulaire> getById(@PathVariable Long id) {
        return ResponseEntity.ok(service.findById(id));
    }
    
    @PostMapping
    public ResponseEntity<Formulaire> create(@RequestBody Formulaire formulaire) {
        Formulaire saved = service.create(formulaire);
        return ResponseEntity.status(HttpStatus.CREATED).body(saved);
    }
    
    @PutMapping("/{id}")
    public ResponseEntity<Formulaire> update(@PathVariable Long id, @RequestBody Formulaire formulaire) {
        return ResponseEntity.ok(service.update(id, formulaire));
    }
    
    @DeleteMapping("/{id}")
    public ResponseEntity<Void> delete(@PathVariable Long id) {
        service.delete(id);
        return ResponseEntity.noContent().build();
    }
    
    @GetMapping("/search")
    public ResponseEntity<List<Formulaire>> search(
        @RequestParam(required = false) String query,
        @RequestParam(required = false) String statut,
        @RequestParam(required = false) String region
    ) {
        return ResponseEntity.ok(service.search(query, statut, region));
    }
}
```

AdminController.java
```java
@RestController
@RequestMapping("/api/admin")
@CrossOrigin(origins = "http://localhost:5173")
public class AdminController {
    
    @Autowired
    private AdminService adminService;
    
    @Autowired
    private JwtTokenProvider tokenProvider;
    
    @PostMapping("/login")
    public ResponseEntity<?> login(@RequestBody LoginRequest request) {
        try {
            Admin admin = adminService.authenticate(request.getUsername(), request.getPassword());
            String token = tokenProvider.generateToken(admin);
            
            return ResponseEntity.ok(new LoginResponse(token, admin));
        } catch (AuthenticationException e) {
            return ResponseEntity.status(HttpStatus.UNAUTHORIZED)
                .body("Identifiants incorrects");
        }
    }
}
```
